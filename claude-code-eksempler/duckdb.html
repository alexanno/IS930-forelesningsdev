        gk.append(box(x0-6000+ix*3600, y0-6000+iy*4000, x0-6000+(ix+1)*3600, y0-6000+(iy+1)*4000))
gk = gpd.GeoDataFrame({"grunnkretsnummer": [f"{42040000+i}" for i in range(len(gk))]}, geometry=gk, crs=crs)

# bygg (punkter)
pts = [Point(x0-6000+rng.random()*18000, y0-6000+rng.random()*12000) for _ in range(500)]

# vassdrag
vass = gpd.GeoDataFrame({"navn": ["Otra", "Bekk"]}, geometry=[LineString([(x0-5000, y0-5500), (x0, y0), (x0+6000, y0+5500)]), LineString([(x0+3000, y0-5000), (x0+3500, y0+2000)])], crs=crs)

brann = gpd.GeoDataFrame({"navn": ["Brannstasjon A", "Brannstasjon B", "Brannstasjon C"]}, geometry=[Point(x0-2000, y0), Point(x0+5000, y0+3000), Point(x0+9000, y0-3000)], crs=crs)

cat > oppgaver.sql <<'EOF'
INSTALL spatial; LOAD spatial;

CREATE OR REPLACE VIEW bygg AS SELECT * FROM ST_Read('bygg.gpkg');
CREATE OR REPLACE VIEW brann AS SELECT * FROM ST_Read('brannstasjon.gpkg');
CREATE OR REPLACE VIEW eiendom AS SELECT * FROM ST_Read('eiendom.gpkg');
CREATE OR REPLACE VIEW kommune AS SELECT * FROM ST_Read('kommuner.gpkg');

.print '--- 1. Bygninger innen 100 m fra brannstasjon'
SELECT b.bygg_id, s.navn AS brannstasjon, round(ST_Distance(b.geom, s.geom), 1) AS avstand_m
FROM bygg b JOIN brann s ON ST_DWithin(b.geom, s.geom, 100)
ORDER BY avstand_m;

.print '--- 2. Brannstasjoner i eiendom'
SELECT s.navn, e.gnr_bnr
FROM brann s JOIN eiendom e ON ST_Intersects(e.geom, s.geom);

.print '--- 3. Eiendommer per kommune'
SELECT k.kommunenavn, count(e.gnr_bnr) AS antall_eiendommer
FROM kommune k LEFT JOIN eiendom e ON ST_Intersects(k.geom, ST_PointOnSurface(e.geom))
GROUP BY k.kommunenavn ORDER BY 1;

.print '--- 4a. Bygninger per kommune'
SELECT k.kommunenavn, count(b.bygg_id) AS antall_bygg
FROM kommune k LEFT JOIN bygg b ON ST_Intersects(k.geom, b.geom)
GROUP BY k.kommunenavn ORDER BY 1;

.print '--- 4b. Bygninger per eiendom (kun eiendommer med bygg)'
SELECT e.gnr_bnr, count(*) AS antall_bygg
FROM eiendom e JOIN bygg b ON ST_Intersects(e.geom, b.geom)
GROUP BY e.gnr_bnr ORDER BY antall_bygg DESC, e.gnr_bnr;
EOF

Alle fire operasjonene kjører på geopackage-filene i eksempeldata/. Jeg brukte DuckDB med spatial-utvidelsen, kjørt via uv run --with duckdb python kjor.py. Jeg har ikke lagt til noe i systemet.

Spørringene ligger i oppgaver.sql, og kjor.py kjører dem og skriver ut resultatene. Alle lagene er i EPSG:25833, så avstander og buffer er i meter uten reprojisering.

Resultater

  1. Bygninger innen 100 m fra brannstasjon: Det er bare én, bygg 318, som ligger 91,5 m fra Brannstasjon A. Spørringen bruker ST_DWithin(b.geom, s.geom, 100).
  2. Brannstasjoner i eiendom: Brannstasjon A ligger i både eiendom 4/1 og 5/1. Stasjonen ligger nøyaktig på grensen mellom de to, så ST_Intersects gir treff på begge. ST_Within eller ST_Contains hadde gitt null treff, fordi de ikke teller punkt på kanten.
  3. Eiendommer per kommune: Kristiansand har 100 eiendommer og Vennesla har 0. Jeg bruker ST_PointOnSurface på eiendommen, slik at en eiendom som krysser en kommunegrense ikke telles to ganger.
  4. Bygninger per kommune og per eiendom: Kristiansand har 286 bygninger og Vennesla 214. Bare 34 av de 100 eiendommene har bygg. Eiendommene 35/1, 48/1 og 75/1 har to hver, og resten har ett. Eiendommer uten bygg er ikke med i listen. Bytt til LEFT JOIN for å få dem med og null som antall.

Bygningene er punkter. Hvis du får ekte byggpolygoner senere, bør oppgave 1 bruke polygonavstanden, og oppgave 4 bør bruke ST_PointOnSurface på polygonen, på samme måte som i oppgave 3.

"""Henter restauranter fra siste Overture Maps-release (via STAC) innenfor kommune-bbox til GeoParquet."""
import json, urllib.request
import duckdb

STAC = "https://stac.overturemaps.org"
get = lambda u: json.load(urllib.request.urlopen(u))

con = duckdb.connect()
con.execute("INSTALL spatial; LOAD spatial; INSTALL httpfs; LOAD httpfs; SET s3_region='us-west-2';")

# 1. bbox i WGS84 fra kommunelaget (EPSG:25833)
xmin, ymin, xmax, ymax = con.sql("""
    WITH b AS (SELECT ST_Extent(ST_Transform(geom, 'EPSG:25833', 'EPSG:4326', always_xy := true)) AS e
               FROM ST_Read('kommuner.gpkg'))
    SELECT ST_XMin(e), ST_YMin(e), ST_XMax(e), ST_YMax(e) FROM b""").fetchone()
print("bbox:", xmin, ymin, xmax, ymax)

# 2. siste release fra STAC, og kun filene (items) som overlapper bbox
release = get(f"{STAC}/catalog.json")["latest"]
print("release:", release)
coll = get(f"{STAC}/{release}/places/place/collection.json")
files = []
for l in coll["links"]:
    if l["rel"] != "item":
        continue
    item = get(l["href"])
    bx0, by0, bx1, by1 = item["bbox"]
    if bx0 <= xmax and bx1 >= xmin and by0 <= ymax and by1 >= ymin:
        files.append(item["assets"]["aws"]["alternate"]["s3"]["href"])
print("filer som overlapper bbox:", len(files))

# 3. Les kun bbox + kategori (bbox-struct gir row group-pruning), skriv GeoParquet
con.execute(f"""
COPY (
  SELECT id,
         names.primary AS navn,
         categories.primary AS kategori,
         basic_category,
         confidence,
         addresses[1].freeform AS adresse,
         addresses[1].locality AS sted,
         geometry
  FROM read_parquet({files}, hive_partitioning=1)
  WHERE bbox.xmin >= {xmin} AND bbox.xmax <= {xmax}
    AND bbox.ymin >= {ymin} AND bbox.ymax <= {ymax}
    AND basic_category = 'restaurant'
) TO 'restauranter.parquet' (FORMAT PARQUET, COMPRESSION ZSTD);
""")
print(con.sql("SELECT count(*) AS antall FROM 'restauranter.parquet'"))

Jeg hentet 179 restauranter fra Overture Maps og lagret dem i restauranter.parquet. Filen er på 12,6 kB og har GeoParquet-metadata (geometrikolonnen geometry, WKB-punkter). GDAL åpner den som et punktlag.

Scriptet hent_restauranter.py gjør dette:

  1. Bbox: Den tar utstrekningen til kommuner.gpkg og transformerer den fra EPSG:25833 til WGS84, som gir 7,888, 58,091, 8,078, 58,207.
  2. Release: catalog.json på stac.overturemaps.org peker på siste release, som er 2026-10-21.0. Scriptet henter bare places/place-collectionen derfra.
  3. Filvalg: Av alle parquet-filene i collectionen leser scriptet bare dem der item-bbox overlapper søkeområdet. Det ble 1 fil.
  4. Filter: DuckDB leser fra S3 og filtrerer på bbox.xmin/xmax/ymin/ymax før noe annet. Da hopper den over de row groups som ligger utenfor, i stedet for å laste hele filen på 730 MB. Deretter filtrerer den på basic_category = 'restaurant'.

Kolonnene i filen er id, navn, kategori, basic_category, confidence, adresse, sted og geometry. Vanligste kategorier er restaurant (77), pizza_restaurant (23) og sushi_restaurant (11).

I denne releasen heter kategorifeltet taxonomy.primary. categories.primary finnes ikke lenger, så eldre eksempler fra Overture feiler her.

Geometrien er i WGS84 (EPSG:4326), mens de andre lagene dine er i EPSG:25833. Før du kobler dem må du enten transformere med ST_Transform(geometry, 'EPSG:4326', 'EPSG:25833', always_xy := true) eller reprojisere ved eksport.

Bbox-en er for hele kommunelaget og ikke bare Kristiansand og Vennesla i seg selv. Hvis du vil ha restauranter helt nøyaktig innenfor kommunegrensene, kan jeg legge til et ST_Intersects-filter mot kommunepolygonene.
