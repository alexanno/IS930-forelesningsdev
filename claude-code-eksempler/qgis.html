import geopandas as gpd, numpy as np
from shapely.geometry import Point, LineString, box
from pyproj import Transformer
rng = np.random.default_rng(1)
tr = Transformer.from_crs(4326, 25833, always_xy=True)
x0, y0 = tr.transform(8.0, 58.15)       # Kristiansand
crs = "EPSG:25833"
# kommuner: to ruter side om side
kommuner = gpd.GeoDataFrame({"kommunenavn": ["Kristiansand", "Vennesla"], "kommunenummer": ["4204", "4223"]},
    geometry=[box(x0-6000, y0-6000, x0+4000, y0+6000), box(x0+4000, y0-6000, x0+12000, y0+6000)], crs=crs)
# grunnkretser: 4x3 ruter
gk = []
for ix in range(5):
    for iy in range(3):
        gk.append(box(x0-6000+ix*3600, y0-6000+iy*4000, x0-6000+(ix+1)*3600, y0-6000+(iy+1)*4000))
gk = gpd.GeoDataFrame({"grunnkretsnummer": [f"{42040000+i}" for i in range(len(gk))]}, geometry=gk, crs=crs)
gk["kommunenummer"] = np.where(gk.geometry.centroid.x < x0+4000, "4204", "4223")
# bygg (punkter)
pts = [Point(x0-6000+rng.random()*18000, y0-6000+rng.random()*12000) for _ in range(500)]
bygg = gpd.GeoDataFrame({"bygg_id": range(500), "bruksareal": rng.integers(30, 400, 500)}, geometry=pts, crs=crs)
# vassdrag
vass = gpd.GeoDataFrame({"navn": ["Otra", "Bekk"]}, geometry=[LineString([(x0-5000, y0-5500), (x0, y0), (x0+6000, y0+5500)]), LineString([(x0+3000, y0-5000), (x0+3500, y0+2000)])], crs=crs)
# brannstasjoner
brann = gpd.GeoDataFrame({"navn": ["Brannstasjon A", "Brannstasjon B", "Brannstasjon C"]}, geometry=[Point(x0-2000, y0), Point(x0+5000, y0+3000), Point(x0+9000, y0-3000)], crs=crs)
# eiendommer (rutenett av små polygoner) og faresone
ei = [box(x0-2000+i*400, y0-2000+j*400, x0-2000+(i+1)*400, y0-2000+(j+1)*400) for i in range(10) for j in range(10)]
eiendom = gpd.GeoDataFrame({"gnr_bnr": [f"{i}/1" for i in range(100)]}, geometry=ei, crs=crs)
fare = gpd.GeoDataFrame({"fare": ["flom"]}, geometry=[vass.geometry.iloc[0].buffer(500)], crs=crs)
for n, g in dict(kommuner=kommuner, grunnkrets=gk, bygg=bygg, vassdrag=vass, brannstasjon=brann, eiendom=eiendom, faresone=fare).items():
    g.to_file(f"{n}.gpkg", layer=n, driver="GPKG")
print("ok", {n: len(g) for n, g in dict(kommuner=kommuner, grunnkrets=gk, bygg=bygg).items()})
"""Henter restauranter fra siste Overture Maps-release (via STAC) innenfor kommune-bbox til GeoParquet."""
import json, urllib.request
import duckdb

STAC = "https://stac.overturemaps.org"
get = lambda u: json.load(urllib.request.urlopen(u))

con = duckdb.connect()
con.execute("INSTALL spatial; LOAD spatial; INSTALL httpfs; LOAD httpfs; SET s3_region='us-west-2';")

# 1. bbox i WGS84 fra kommunelaget (EPSG:25833)
xmin, ymin, xmax, ymax = con.sql("""
    WITH b AS (SELECT ST_Extent(ST_Transform(geom, 'EPSG:25833', 'EPSG:4326', always_xy := true)) AS e
               FROM ST_Read('kommuner.gpkg'))
    SELECT ST_XMin(e), ST_YMin(e), ST_XMax(e), ST_YMax(e) FROM b""").fetchone()
print("bbox:", xmin, ymin, xmax, ymax)

# 2. siste release fra STAC, og kun filene (items) som overlapper bbox
release = get(f"{STAC}/catalog.json")["latest"]
print("release:", release)
coll = get(f"{STAC}/{release}/places/place/collection.json")
files = []
for l in coll["links"]:
    if l["rel"] != "item":
        continue
    item = get(l["href"])
    bx0, by0, bx1, by1 = item["bbox"]
    if bx0 <= xmax and bx1 >= xmin and by0 <= ymax and by1 >= ymin:
        files.append(item["assets"]["aws"]["alternate"]["s3"]["href"])
print("filer som overlapper bbox:", len(files))

# 3. Les kun bbox + kategori (bbox-struct gir row group-pruning), skriv GeoParquet
con.execute(f"""
COPY (
  SELECT id,
         names.primary AS navn,
         taxonomy.primary AS kategori,
         basic_category,
         confidence,
         addresses[1].freeform AS adresse,
         addresses[1].locality AS sted,
         geometry
  FROM read_parquet({files}, hive_partitioning=1)
  WHERE bbox.xmin >= {xmin} AND bbox.xmax <= {xmax}
    AND bbox.ymin >= {ymin} AND bbox.ymax <= {ymax}
    AND basic_category = 'restaurant'
) TO 'restauranter.parquet' (FORMAT PARQUET,
Vis resten (5,814 tegn til)
 COMPRESSION ZSTD);
""")
print(con.sql("SELECT count(*) AS antall FROM 'restauranter.parquet'"))

Jeg har laget QGIS-prosjektet eksempeldata.qgz med alle datasettene og tetthetslaget. Jeg har bare sett det som et bilde rendret fra skriptet, ikke åpnet det i selve QGIS. Bakgrunnskartet (CARTO Positron) lastet ikke i den rendringen, så det er uprøvd.

Tetthetslaget heter «Restauranttetthet per eiendom». Det er lagret i restauranttetthet.gpkg, laget eiendom_restauranttetthet. For hver eiendom har det antall restauranter, areal i km² og restauranter per km². Av 179 restauranter ligger 146 innenfor eiendommene, og 75 av 100 eiendommer har ingen. Sentrum skiller seg ut: én eiendom har 33 restauranter, og de neste har 25, 24, 15 og 14.

  • Klasser og farger: Fargene går i gult til mørkerødt: 0, 1, 2–3, 4–15, 16–30 og over 30 restauranter. Jeg valgte klassene ut fra fordelingen, siden jevne klasser ville gjort nesten alle eiendommer like. Antallet vises som tall på eiendommene, og musepeker-info viser antall og tetthet per km².
  • Heatmap: Lagtreet har også et heatmap av restaurantpunktene, skrudd av som standard.

De øvrige lagene, ordnet i grupper i lagtreet:

  • Kommuner: pastellflater med stiplet grense og navn.
  • Grunnkretser: tynne grå linjer med nummer.
  • Faresone (flom): gjennomsiktig blå flate.
  • Buffer (DuckDB): rød stiplet kontur.
  • Eiendommer: kildelaget, skjult som standard fordi tetthetslaget dekker det.
  • Vassdrag: blå linjer, med Otra tykkere enn bekken.
  • Bygg: grå prikker som er større jo større bruksarealet er.
  • Brannstasjoner: røde kvadrater med navn.
  • Restauranter: oransje prikker, lest direkte fra restauranter.parquet (trenger Parquet-støtte i QGIS).
  • Bakgrunnskart: CARTO Positron nederst, som krever nett.

Prosjektet bruker EPSG:25833 og relative stier, og har bokmerkene «Hele området» og «Eiendommer – restauranttetthet».

Jeg bygde alt med PyQGIS 3.44 i skriptet lag_qgis_prosjekt.py. Kjørekommandoen står øverst i filen, og du kan endre farger og klassegrenser der og kjøre det på nytt.
